for file in \
    /var/log/vsftpd.log \
    /var/log/xferlog \
    /var/log/proftpd/proftpd.log \
    /var/log/auth.log \
    /var/log/syslog
do
    if [ -r "$file" ]; then
        echo "===== $file ====="

        grep -Ei \
        "ftp|vsftpd|login|anonymous|upload|download|connect|failed|success|FLAG\\{" \
        "$file" 2>/dev/null |
        tail -n 1000

        echo
    fi
done
