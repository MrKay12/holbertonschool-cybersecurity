    /etc/logicorp \
    /opt/logicorp \
    /srv/ftp \
    /var/lib/logicorp \
    /usr/local/logicorp
do
    if [ -e "$directory" ]; then
        echo "===== $directory ====="
        find "$directory" \
            -maxdepth 6 \
            -printf "%M %u:%g %s %p\n" 2>/dev/null
        echo
    fi
done
