    /etc/logicorp \
    /opt/logicorp \
    /srv/ftp \
    /var/lib/logicorp \
    /usr/local/logicorp
do
    [ -d "$directory" ] || continue

    find "$directory" \
        -maxdepth 6 \
        -type f \
        -readable \
        -size -5M \
        -print \
        -exec sh -c '
            echo "===== $1 ====="
            sed -n "1,500p" "$1"
            echo
        ' sh {} \; 2>/dev/null
done
