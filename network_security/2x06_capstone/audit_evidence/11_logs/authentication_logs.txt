for file in \
    /var/log/auth.log \
    /var/log/secure \
    /var/log/syslog \
    /var/log/messages
do
    if [ -r "$file" ]; then
        echo "===== $file ====="

        grep -Ei \
        "accepted|failed|invalid user|root|sudo|authentication failure|session opened|session closed|blocked|ban|unban|FLAG\{" \
        "$file" 2>/dev/null |
        tail -n 500

        echo
    fi
done
